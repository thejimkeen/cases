# RPG 职业教育系统 - 开发路径与数据库设计

## 三、数据库设计

### 3.1 关系型数据库（PostgreSQL）

#### 用户表 (users)

```sql
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  username VARCHAR(50) UNIQUE NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role VARCHAR(20) NOT NULL, -- 'student', 'teacher', 'admin'
  real_name VARCHAR(100),
  student_id VARCHAR(50), -- 学号
  class_id INT REFERENCES classes(id),
  avatar_url VARCHAR(255),
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  last_login_at TIMESTAMP
);

CREATE INDEX idx_users_class ON users(class_id);
CREATE INDEX idx_users_role ON users(role);
```

#### 班级表 (classes)

```sql
CREATE TABLE classes (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  code VARCHAR(50) UNIQUE NOT NULL, -- 班级代码
  teacher_id INT REFERENCES users(id),
  academic_year VARCHAR(20), -- '2024-2025'
  semester INT, -- 1 或 2
  created_at TIMESTAMP DEFAULT NOW()
);
```

#### 课程表 (courses)

```sql
CREATE TABLE courses (
  id SERIAL PRIMARY KEY,
  title VARCHAR(200) NOT NULL,
  description TEXT,
  category VARCHAR(50), -- 'logistics', 'supply_chain'
  difficulty INT, -- 1-5
  estimated_minutes INT,
  cover_image VARCHAR(255),
  created_by INT REFERENCES users(id),
  status VARCHAR(20) DEFAULT 'draft', -- 'draft', 'published', 'archived'
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

#### 场景表 (scenes)

```sql
CREATE TABLE scenes (
  id SERIAL PRIMARY KEY,
  course_id INT REFERENCES courses(id),
  scene_key VARCHAR(100) UNIQUE NOT NULL, -- 'scene_001'
  title VARCHAR(200) NOT NULL,
  description TEXT,
  difficulty INT,
  estimated_minutes INT,
  order_index INT, -- 在课程中的顺序
  scene_data JSONB NOT NULL, -- 完整的场景定义 JSON
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_scenes_course ON scenes(course_id);
CREATE INDEX idx_scenes_key ON scenes(scene_key);
```

#### 学生进度表 (student_progress)

```sql
CREATE TABLE student_progress (
  id SERIAL PRIMARY KEY,
  student_id INT REFERENCES users(id),
  scene_id INT REFERENCES scenes(id),
  status VARCHAR(20) NOT NULL, -- 'not_started', 'in_progress', 'completed'
  current_node_id VARCHAR(100),
  game_state JSONB, -- 当前游戏状态
  decisions JSONB[], -- 决策历史数组
  started_at TIMESTAMP,
  completed_at TIMESTAMP,
  time_spent_seconds INT DEFAULT 0,
  attempt_count INT DEFAULT 1,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(student_id, scene_id, attempt_count)
);

CREATE INDEX idx_progress_student ON student_progress(student_id);
CREATE INDEX idx_progress_scene ON student_progress(scene_id);
CREATE INDEX idx_progress_status ON student_progress(status);
```

#### 成绩表 (scores)

```sql
CREATE TABLE scores (
  id SERIAL PRIMARY KEY,
  progress_id INT REFERENCES student_progress(id),
  student_id INT REFERENCES users(id),
  scene_id INT REFERENCES scenes(id),
  total_score DECIMAL(5,2), -- 总分（0-100）
  grade VARCHAR(2), -- 'S', 'A', 'B', 'C', 'D'
  dimension_scores JSONB, -- 各维度得分
  feedback JSONB, -- 反馈内容
  achievements JSONB[], -- 获得的成就
  npc_reactions JSONB, -- NPC 反应
  scored_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_scores_student ON scores(student_id);
CREATE INDEX idx_scores_scene ON scores(scene_id);
CREATE INDEX idx_scores_grade ON scores(grade);
```

#### 成就表 (achievements)

```sql
CREATE TABLE achievements (
  id SERIAL PRIMARY KEY,
  key VARCHAR(100) UNIQUE NOT NULL,
  name VARCHAR(100) NOT NULL,
  description TEXT,
  icon VARCHAR(255),
  category VARCHAR(50), -- 'info_gathering', 'decision_making'
  rarity VARCHAR(20), -- 'common', 'rare', 'epic', 'legendary'
  points INT DEFAULT 0
);

-- 学生成就关联表
CREATE TABLE student_achievements (
  id SERIAL PRIMARY KEY,
  student_id INT REFERENCES users(id),
  achievement_id INT REFERENCES achievements(id),
  scene_id INT REFERENCES scenes(id),
  unlocked_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(student_id, achievement_id)
);
```

#### 决策分析表 (decision_analytics)

```sql
CREATE TABLE decision_analytics (
  id SERIAL PRIMARY KEY,
  scene_id INT REFERENCES scenes(id),
  node_id VARCHAR(100) NOT NULL,
  option_id VARCHAR(10) NOT NULL,
  student_id INT REFERENCES users(id),
  progress_id INT REFERENCES student_progress(id),
  decision_time_seconds INT, -- 决策用时
  is_optimal BOOLEAN, -- 是否是最优选择
  outcome VARCHAR(20), -- 'success', 'failure', 'neutral'
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_analytics_scene_node ON decision_analytics(scene_id, node_id);
CREATE INDEX idx_analytics_student ON decision_analytics(student_id);
```

## 六、技术实现关键代码示例

### 6.1 后端：场景引擎 API

**文件：`backend/src/routes/game.js`**

```javascript
const express = require('express');
const router = express.Router();
const SceneEngine = require('../services/SceneEngine');
const { authenticate } = require('../middlewares/auth');

// 开始场景
router.post('/scenes/:sceneKey/start', authenticate, async (req, res) => {
  try {
    const { sceneKey } = req.params;
    const userId = req.user.id;

    // 获取场景定义
    const sceneData = await SceneModel.findByKey(sceneKey);
    if (!sceneData) {
      return res.status(404).json({
        success: false,
        error: { message: '场景不存在' }
      });
    }

    // 创建新的进度记录
    const progress = await Progress.create({
      studentId: userId,
      sceneId: sceneData.id,
      status: 'in_progress',
      currentNodeId: sceneData.scene_data.startNodeId,
      gameState: sceneData.scene_data.globalState,
      decisions: [],
      startedAt: new Date()
    });

    // 初始化引擎
    const engine = new SceneEngine(sceneData.scene_data, {
      studentId: userId,
      progressId: progress.id
    });

    // 保存到 MongoDB（热数据）
    await SessionModel.create({
      studentId: userId,
      sceneKey,
      progressId: progress.id,
      sessionState: engine.state
    });

    // 返回初始节点
    const firstNode = engine.getCurrentNode();

    res.json({
      success: true,
      data: {
        progressId: progress.id,
        currentNode: firstNode,
        gameState: engine.state.globalState
      }
    });

  } catch (error) {
    console.error('Start scene error:', error);
    res.status(500).json({
      success: false,
      error: { message: '启动场景失败' }
    });
  }
});

// 提交决策
router.post('/game/decision', authenticate, async (req, res) => {
  try {
    const { progressId, nodeId, optionId } = req.body;
    const userId = req.user.id;

    // 验证进度归属
    const progress = await Progress.findById(progressId);
    if (!progress || progress.studentId !== userId) {
      return res.status(403).json({
        success: false,
        error: { message: '无权操作该进度' }
      });
    }

    // 加载引擎
    const engine = await SceneEngine.loadProgress(progressId, userId);

    // 处理决策
    const result = await engine.makeChoice(optionId);

    if (!result.success) {
      return res.status(400).json({
        success: false,
        error: { message: result.message }
      });
    }

    // 记录决策分析
    await DecisionAnalytics.create({
      sceneId: progress.sceneId,
      nodeId,
      optionId,
      studentId: userId,
      progressId,
      decisionTimeSeconds: calculateDecisionTime(req.body.startTime),
      isOptimal: checkIfOptimal(nodeId, optionId),
      outcome: result.outcome || 'neutral'
    });

    // 检查是否完成场景
    const isComplete = engine.isSceneComplete();
    let scoreData = null;

    if (isComplete) {
      // 计算评分
      scoreData = engine.calculateScore();
      
      // 保存成绩
      await Score.create({
        progressId,
        studentId: userId,
        sceneId: progress.sceneId,
        totalScore: scoreData.totalScore,
        grade: scoreData.grade,
        dimensionScores: scoreData.dimensionScores,
        feedback: scoreData.feedback,
        achievements: scoreData.achievements,
        npcReactions: scoreData.npcReactions
      });

      // 更新进度状态
      await Progress.update(progressId, {
        status: 'completed',
        completedAt: new Date(),
        timeSpentSeconds: calculateTotalTime(progress.startedAt)
      });

      // 解锁成就
      await unlockAchievements(userId, scoreData.achievements);
    }

    res.json({
      success: true,
      data: {
        nextNode: result.nextNode,
        stateChanges: result.stateChanges,
        isComplete,
        score: scoreData
      }
    });

  } catch (error) {
    console.error('Decision error:', error);
    res.status(500).json({
      success: false,
      error: { message: '处理决策失败' }
    });
  }
});

// 提交表单
router.post('/game/form-submit', authenticate, async (req, res) => {
  try {
    const { progressId, formId, formData } = req.body;
    const userId = req.user.id;

    // 加载引擎
    const engine = await SceneEngine.loadProgress(progressId, userId);
    const currentNode = engine.getCurrentNode();

    // 验证表单
    const validation = validateForm(currentNode.form, formData);
    if (!validation.valid) {
      return res.status(400).json({
        success: false,
        error: {
          message: '表单验证失败',
          details: validation.errors
        }
      });
    }

    // 计算表单得分
    const formScore = calculateFormScore(currentNode.form, formData, engine.state);

    // 更新状态
    engine.state.globalState.formData = formData;
    engine.state.globalState.formScore = formScore;

    // 跳转到下一个节点
    const nextNodeId = currentNode.form.onSubmit.nextNodeId;
    engine.state.currentNodeId = nextNodeId;

    await engine.saveProgress();

    res.json({
      success: true,
      data: {
        formScore,
        feedback: generateFormFeedback(formData, formScore),
        nextNode: engine.getCurrentNode()
      }
    });

  } catch (error) {
    console.error('Form submit error:', error);
    res.status(500).json({
      success: false,
      error: { message: '提交表单失败' }
    });
  }
});

module.exports = router;
```

        <div class="dialogue-text" v-html="formatDialogue(currentNode.dialogue.text)"></div>
        
        <!-- 打字机效果控制 -->
        <div class="dialogue-controls">
          <button @click="skipTyping" v-if="isTyping">跳过 ⏩</button>
          <button @click="nextDialogue" v-else>继续 →</button>
        </div>
      </div>

      <!-- 旁白 -->
      <div class="narration" v-if="currentNode.narration">
        <div class="narration-text" v-html="currentNode.narration"></div>
      </div>

      <!-- 选项 -->
      <div class="options-container" v-if="showOptions && currentNode.options">
        <div 
          class="option-button"
          v-for="option in currentNode.options"
          :key="option.id"
          @click="makeChoice(option)"
          :class="{ 'disabled': !canChoose(option) }"
        >
          <div class="option-label">[{{ option.id }}]</div>
          <div class="option-text">{{ option.text }}</div>
          <div class="option-hint" v-if="option.hint">{{ option.hint }}</div>
          <div class="option-meta">
            <span v-if="option.timeRequired">⏱️ {{ option.timeRequired }} 分钟</span>
          </div>
        </div>
      </div>

      <!-- 表单节点 -->
      <TransportForm 
        v-if="currentNode.type === 'form'"
        :game-state="gameState"
        :on-submit="handleFormSubmit"
      />

      <!-- 计算器节点 -->
      <CostCalculator 
        v-if="currentNode.type === 'calculator'"
        :calculator-config="currentNode.calculator"
        :on-calculate="handleCalculation"
      />
    </div>

    <!-- 反馈提示 -->
    <transition name="fade">
      <div class="feedback-toast" v-if="feedback">
        <div class="feedback-content" :class="feedback.type">
          {{ feedback.message }}
          <div class="feedback-effects" v-if="feedback.effects">
            <span v-for="(value, key) in feedback.effects" :key="key" class="effect-item">
              {{ formatEffect(key, value) }}
            </span>
          </div>
        </div>
      </div>
    </transition>

    <!-- 加载中 -->
    <div class="loading-overlay" v-if="loading">
      <div class="spinner"></div>
      <p>正在加载...</p>
    </div>
  </div>
</template>

<script>
import { ref, computed, onMounted, watch } from 'vue';
import { useRoute } from 'vue-router';
import { gameAPI } from '@/api/game';
import TransportForm from './tools/TransportForm.vue';
import CostCalculator from './tools/CostCalculator.vue';

export default {
  name: 'SceneGame',
  components: {
    TransportForm,
    CostCalculator
  },
  setup() {
    const route = useRoute();
    const sceneKey = route.params.sceneKey;

    // 状态
    const loading = ref(false);
    const progressId = ref(null);
    const currentNode = ref(null);
    const gameState = ref({});
    const sceneData = ref(null);
    const isTyping = ref(false);
    const showOptions = ref(false);
    const feedback = ref(null);

    // 背景样式
    const backgroundStyle = computed(() => {
      if (!currentNode.value?.background) return {};
      return {
        backgroundImage: `url(${currentNode.value.background})`
      };
    });

    // 初始化场景
    const initScene = async () => {
      loading.value = true;
      try {
        const response = await gameAPI.startScene(sceneKey);
        progressId.value = response.data.progressId;
        currentNode.value = response.data.currentNode;
        gameState.value = response.data.gameState;
        sceneData.value = response.data.sceneData;

        // 开始打字机效果
        if (currentNode.value.type === 'dialogue') {
          startTypingEffect();
        }
      } catch (error) {
        console.error('Init scene failed:', error);
        alert('加载场景失败');
      } finally {
        loading.value = false;
      }
    };

    // 打字机效果
    const startTypingEffect = () => {
      isTyping.value = true;
      showOptions.value = false;
      
      // 模拟打字效果（实际可以使用更复杂的实现）
      setTimeout(() => {
        isTyping.value = false;
        showOptions.value = true;
      }, currentNode.value.dialogue.text.length * 30); // 每字30ms
    };

    const skipTyping = () => {
      isTyping.value = false;
      showOptions.value = true;
    };

    const nextDialogue = () => {
      // 如果没有选项，自动跳转到下一个节点
      if (!currentNode.value.options || currentNode.value.options.length === 0) {
        const nextNodeId = currentNode.value.nextNodeId;
        if (nextNodeId) {
          // 加载下一个节点
          loadNode(nextNodeId);
        }
      }
    };

    // 做出选择
    const makeChoice = async (option) => {
      if (!canChoose(option)) return;

      loading.value = true;
      try {
        const response = await gameAPI.makeDecision({
          progressId: progressId.value,
          nodeId: currentNode.value.nodeId,
          optionId: option.id
        });

        // 显示反馈
        if (response.data.stateChanges) {
          showFeedback({
            type: 'success',
            message: '✓ 决策已记录',
            effects: response.data.stateChanges
          });
        }

        // 更新游戏状态
        gameState.value = response.data.gameState;

        // 检查是否完成
        if (response.data.isComplete) {
          showScoreScreen(response.data.score);
        } else {
          // 加载下一个节点
          currentNode.value = response.data.nextNode;
          if (currentNode.value.type === 'dialogue') {
            startTypingEffect();
          }
        }

      } catch (error) {
        console.error('Make choice failed:', error);
        showFeedback({
          type: 'error',
          message: error.response?.data?.error?.message || '操作失败'
        });
      } finally {
        loading.value = false;
      }
    };

    // 检查是否可以选择
    const canChoose = (option) => {
      if (!option.conditions) return true;
      // 评估条件
      return evaluateCondition(option.conditions, gameState.value);
    };

    // 显示反馈
    const showFeedback = (feedbackData) => {
      feedback.value = feedbackData;
      setTimeout(() => {
        feedback.value = null;
      }, 3000);
    };

    // 格式化效果显示
    const formatEffect = (key, value) => {
      const labels = {
        'experience.infoGathering': '信息收集',
        'trustZhangLei': '张雷信任度',
        'infoCompleteness': '信息完整度'
      };
      const label = labels[key] || key;
      const prefix = value > 0 ? '+' : '';
      return `${label} ${prefix}${value}`;
    };

    // 处理表单提交
    const handleFormSubmit = async (formData) => {
      loading.value = true;
      try {
        const response = await gameAPI.submitForm({
          progressId: progressId.value,
          formId: currentNode.value.form.formId,
          formData
        });

        showFeedback({
          type: 'success',
          message: response.data.feedback,
          score: response.data.formScore
        });

        // 跳转到下一个节点
        currentNode.value = response.data.nextNode;

      } catch (error) {
        console.error('Form submit failed:', error);
        showFeedback({
          type: 'error',
          message: error.response?.data?.error?.message || '提交失败'
        });
      } finally {
        loading.value = false;
      }
    };

    // 获取 NPC 信息
    const getNPCAvatar = (npcId) => {
      const npc = sceneData.value?.npcs?.find(n => n.id === npcId);
      return npc?.avatar || '/assets/npcs/default.png';
    };

    const getNPCName = (npcId) => {
      const npc = sceneData.value?.npcs?.find(n => n.id === npcId);
      return npc?.name || npcId;
    };

    // 格式化对话文本
    const formatDialogue = (text) => {
      // 支持简单的 markdown 样式
      return text
        .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
        .replace(/\n/g, '<br/>');
    };

    // 显示评分界面
    const showScoreScreen = (scoreData) => {
      // 跳转到评分页面
      router.push({
        name: 'SceneScore',
        params: { progressId: progressId.value },
        query: { score: JSON.stringify(scoreData) }
      });
    };

    // 初始化
    onMounted(() => {
      initScene();
    });

    return {
      loading,
      currentNode,
      gameState,
      isTyping,
      showOptions,
      feedback,
      backgroundStyle,
      makeChoice,
      canChoose,
      skipTyping,
      nextDialogue,
      handleFormSubmit,
      getNPCAvatar,
      getNPCName,
      formatDialogue,
      formatEffect
    };
  }
};
</script>
