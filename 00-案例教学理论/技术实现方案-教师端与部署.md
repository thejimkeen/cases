### 7.3 场景编辑器（简化版）

```javascript
// 场景编辑器核心数据结构
const sceneEditor = {
  sceneData: {
    sceneKey: 'scene_002',
    title: '',
    description: '',
    nodes: {},
    npcs: [],
    globalState: {}
  },
  
  currentNodeId: null,
  
  // 添加节点
  addNode(type) {
    const nodeId = `node_${Date.now()}`;
    const newNode = {
      nodeId,
      type, // 'dialogue', 'form', 'calculator'
      title: '新节点',
      ...getNodeTemplate(type)
    };
    
    this.sceneData.nodes[nodeId] = newNode;
    this.currentNodeId = nodeId;
    
    return nodeId;
  },
  
  // 连接节点
  connectNodes(fromNodeId, optionId, toNodeId) {
    const fromNode = this.sceneData.nodes[fromNodeId];
    const option = fromNode.options.find(opt => opt.id === optionId);
    if (option) {
      option.nextNodeId = toNodeId;
    }
  },
  
  // 验证场景完整性
  validate() {
    const errors = [];
    
    // 检查起始节点
    if (!this.sceneData.startNodeId) {
      errors.push('缺少起始节点');
    }
    
    // 检查每个节点
    for (const [nodeId, node] of Object.entries(this.sceneData.nodes)) {
      // 检查选项连接
      if (node.options) {
        for (const option of node.options) {
          if (!option.nextNodeId) {
            errors.push(`节点 ${nodeId} 的选项 ${option.id} 未连接下一个节点`);
          }
        }
      }
      
      // 检查必填字段
      if (!node.title) {
        errors.push(`节点 ${nodeId} 缺少标题`);
      }
    }
    
    // 检查孤立节点
    const reachableNodes = this.getReachableNodes();
    const allNodeIds = Object.keys(this.sceneData.nodes);
    const orphanNodes = allNodeIds.filter(id => !reachableNodes.has(id));
    if (orphanNodes.length > 0) {
      errors.push(`存在孤立节点：${orphanNodes.join(', ')}`);
    }
    
    return {
      valid: errors.length === 0,
      errors
    };
  },
  
  // 获取可达节点
  getReachableNodes() {
    const reachable = new Set();
    const queue = [this.sceneData.startNodeId];
    
    while (queue.length > 0) {
      const nodeId = queue.shift();
      if (reachable.has(nodeId)) continue;
      
      reachable.add(nodeId);
      
      const node = this.sceneData.nodes[nodeId];
      if (node.options) {
        for (const option of node.options) {
          if (option.nextNodeId && !reachable.has(option.nextNodeId)) {
            queue.push(option.nextNodeId);
          }
        }
      }
    }
    
    return reachable;
  },
  
  // 导出场景
  export() {
    const validation = this.validate();
    if (!validation.valid) {
      throw new Error(`场景验证失败：${validation.errors.join('; ')}`);
    }
    
    return JSON.stringify(this.sceneData, null, 2);
  },
  
  // 导入场景
  import(jsonData) {
    try {
      const data = JSON.parse(jsonData);
      this.sceneData = data;
      this.currentNodeId = data.startNodeId;
      return { success: true };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }
};
```

```javascript
// 场景定义缓存（Redis）
async function getSceneDefinition(sceneKey) {
  const cacheKey = `scene:${sceneKey}`;
  
  // 先查缓存
  let sceneData = await redis.get(cacheKey);
  if (sceneData) {
    return JSON.parse(sceneData);
  }
  
  // 从数据库加载
  sceneData = await SceneModel.findByKey(sceneKey);
  
  // 写入缓存（1小时）
  await redis.setex(cacheKey, 3600, JSON.stringify(sceneData));
  
  return sceneData;
}

// 排行榜缓存（Redis ZSET）
async function updateLeaderboard(sceneKey, studentId, score) {
  const key = `leaderboard:${sceneKey}`;
  await redis.zadd(key, score, studentId);
  await redis.expire(key, 86400); // 24小时过期
}

async function getLeaderboard(sceneKey, limit = 10) {
  const key = `leaderboard:${sceneKey}`;
  return await redis.zrevrange(key, 0, limit - 1, 'WITHSCORES');
}
```

```javascript
// JWT 中间件
const jwt = require('jsonwebtoken');

function authenticate(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({ error: 'No token provided' });
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded;
    next();
  } catch (error) {
    return res.status(401).json({ error: 'Invalid token' });
  }
}

// 角色权限检查
function requireRole(...roles) {
  return (req, res, next) => {
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'Insufficient permissions' });
    }
    next();
  };
}

// 使用示例
app.get('/api/teacher/students', 
  authenticate, 
  requireRole('teacher', 'admin'), 
  async (req, res) => {
    // ... 处理逻辑
  }
);
```

```javascript
const Joi = require('joi');

// 定义验证模式
const decisionSchema = Joi.object({

  nodeId: Joi.string().pattern(/^node_\d+$/).required(),
  optionId: Joi.string().max(10).required()

// 验证中间件
function validate(schema) {
  return (req, res, next) => {
    const { error, value } = schema.validate(req.body);
    
    if (error) {
      return res.status(400).json({
        error: 'Validation failed',
        details: error.details.map(d => d.message)
      });
    }
    
    req.validatedData = value;
    next();
  };
}

// 使用
app.post('/api/game/decision', 
  authenticate, 
  validate(decisionSchema),
  async (req, res) => {
    // 使用 req.validatedData
  }
);
```

## 十、总结与后续规划

### 10.1 核心价值总结

这个 RPG 职业教育系统的核心价值在于：

**1. 沉浸式学习体验**
- 将枯燥的案例变成有剧情的游戏
- 学生主动探索，而不是被动接受
- 即时反馈，增强学习动力

**2. 真实场景模拟**
- 填单、计算、查询等真实工作环节
- 决策后果可见，风险成本可控
- 培养职业素养和决策能力

**3. 数据驱动教学**
- 教师可以看到每个学生的决策路径
- 识别常见错误和薄弱环节
- 个性化指导和课程优化

**4. 可扩展架构**
- 模块化设计，易于添加新场景
- 支持多学科应用
- 技术栈成熟，团队易上手

### 10.2 开发优先级建议

**第一阶段（MVP，2-3 周）：**
1. ✅ 完成"货车选择的第一天"场景
2. ✅ 学生端基础功能
3. ✅ 简单的评分系统

**第二阶段（基础版，4-6 周）：**
1. ✅ 填单和计算工具
2. ✅ 排行榜和成就系统
3. ✅ 教师端数据看板

**第三阶段（完整版，7-12 周）：**
1. ✅ 场景编辑器
2. ✅ 高级数据分析
3. ✅ 多场景和章节系统

### 10.3 技术栈推荐（基于讨论）

**最小可行产品（快速验证）：**
- 前端：Vue 3 + Vite + Vant
- 后端：Node.js + Express + Prisma
- 数据库：PostgreSQL + MongoDB + Redis
- 部署：单服务器 + Docker Compose

**理由：**
- Vue 学习曲线平缓，国内资源多
- Node.js 前后端统一语言
- 部署简单，成本低

### 10.4 成本估算

**开发成本（外包）：**
- MVP 版本：3-5 万元（2-3 周）
- 基础版：8-12 万元（4-6 周）
- 完整版：15-25 万元（7-12 周）

**自研成本（人月）：**
- MVP：1 前端 + 1 后端，共 2 人月
- 基础版：1 前端 + 1 后端 + 0.5 UI，共 4 人月
- 完整版：1.5 前端 + 1.5 后端 + 0.5 UI，共 8 人月

**运营成本（月）：**

- CDN/OSS：200-1000 元
